;;; ============================================================
;;; CHKPARENS2 — диагностика баланса скобок LSP-файла
;;; ------------------------------------------------------------
;;; Что делает:
;;;   построчно читает D:/autoextraction/Extraction/cladding.lsp
;;;   и считает баланс скобок: "(" -> +1, ")" -> -1.
;;;
;;; Что игнорирует (чтобы не считать скобки внутри текста):
;;;   - строковые литералы в кавычках "..." , включая
;;;     экранированные кавычки \" (флаг skip пропускает символ
;;;     после обратного слэша);
;;;   - комментарии: от ";" до конца строки (флаг incomm).
;;;
;;; Что печатает:
;;;   "БАЛАНС n ПЕРЕД строкой N: (defun ..." - если перед началом
;;;     очередной функции счётчик не равен 0, значит предыдущая
;;;     функция НЕ закрыта (не хватает закрывающих скобок);
;;;   "ИТОГ: n" - конечный баланс всего файла.
;;;
;;; Норма:
;;;   ИТОГ: 0.
;;;   Допустимо одно сообщение "БАЛАНС 1 ПЕРЕД ... (defun *error*"
;;;   - это вложенная defun внутри cladding-main, ложное
;;;   срабатывание, а не ошибка.
;;;
;;; Как пользоваться:
;;;   1) загрузить этот файл в AutoCAD (или вставить код в командную строку);
;;;   2) выполнить команду CHKPARENS2;
;;;   3) если ИТОГ не 0 - искать скобку в последней функции перед
;;;      первой строкой с сообщением "БАЛАНС ...".
;;;
;;; Ограничения:
;;;   - путь к проверяемому файлу зашит в коде (open ...);
;;;   - не проверяет парность кавычек и квадратные скобки;
;;;   - не понимает многострочные строковые литералы.
;;; ============================================================

(defun c:CHKPARENS2 ( / f ln line bal i ch instr incomm skip tr)
  (setq f (open "D:/autoextraction/Extraction/cladding.lsp" "r"))
  (if (null f)
    (princ "\nНе удалось открыть cladding.lsp.")
    (progn
      (setq ln 0 bal 0)
      (while (setq line (read-line f))
        (setq ln (1+ ln))
        (setq tr (vl-string-trim " \t" line))
        (if (and (= (substr tr 1 6) "(defun") (/= bal 0))
          (princ (strcat "\nБАЛАНС " (itoa bal) " ПЕРЕД строкой " (itoa ln) ": " tr))
        )
        (setq i 1 instr nil incomm nil skip nil)
        (while (<= i (strlen line))
          (setq ch (substr line i 1))
          (cond
            (skip (setq skip nil))
            (incomm nil)
            ((and instr (= ch "\\")) (setq skip T))
            (instr (if (= ch "\"") (setq instr nil)))
            ((= ch ";") (setq incomm T))
            ((= ch "\"") (setq instr T))
            ((= ch "(") (setq bal (1+ bal)))
            ((= ch ")") (setq bal (1- bal)))
          )
          (setq i (1+ i))
        )
      )
      (close f)
      (princ (strcat "\nИТОГ: " (itoa bal)))
    )
  )
  (princ)
)
